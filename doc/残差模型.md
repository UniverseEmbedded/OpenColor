```python
"""
ML残差模型
使用机器学习模型（如XGBoost）学习RTS预测的残差
"""

from __future__ import annotations

import json
import pickle
from dataclasses import dataclass
from pathlib import Path

import numpy as np
from loguru import logger

from oc_core_02.utils.logger import get_logger

logger = get_logger(__name__)


@dataclass
class MLResidualModel:
    """ML残差模型"""
    
    # RTS模型参数（用于基础预测）
    rts_alpha: np.ndarray  # 材料反射率参数 (n_mats, 3)
    rts_beta: np.ndarray   # 材料透射率参数 (n_mats, 3)
    rts_gamma: np.ndarray  # 背板反射率参数 (3,)
    material_keys: list[str]
    n_layers: int
    layer_names_order: str
    
    # ML残差模型（每个Lab通道一个模型）
    model_L: object  # XGBoost/LightGBM模型
    model_a: object
    model_b: object
    
    # 特征名称
    feature_names: list[str]
    
    def predict(self, sequences: list[list[str]], X_features: np.ndarray) -> np.ndarray:
        """
        预测颜色（Lab空间）
        
        Args:
            sequences: 层序列列表，每个序列是材料名列表
            X_features: 特征矩阵 (N, n_features)
            
        Returns:
            pred_lab: 预测颜色 (N, 3)
        """
        # 1. RTS基础预测
        rts_pred_lab = self._rts_predict_lab(sequences)
        
        # 2. ML残差预测
        residual_L = self.model_L.predict(X_features)
        residual_a = self.model_a.predict(X_features)
        residual_b = self.model_b.predict(X_features)
        residual_lab = np.column_stack([residual_L, residual_a, residual_b])
        
        # 3. 融合
        final_pred_lab = rts_pred_lab + residual_lab
        return final_pred_lab
    
    def _rts_predict_lab(self, sequences: list[list[str]]) -> np.ndarray:
        """RTS模型基础预测"""
        # 实现RTS预测逻辑
        pass
    
    def save(self, out_dir: Path) -> dict[str, str]:
        """保存模型"""
        out_dir.mkdir(parents=True, exist_ok=True)
        
        # 保存RTS参数
        rts_params = {
            "alpha": self.rts_alpha.tolist(),
            "beta": self.rts_beta.tolist(),
            "gamma": self.rts_gamma.tolist(),
            "material_keys": self.material_keys,
            "n_layers": self.n_layers,
            "layer_names_order": self.layer_names_order,
            "feature_names": self.feature_names,
        }
        (out_dir / "rts_params.json").write_text(
            json.dumps(rts_params, indent=2), encoding="utf-8"
        )
        
        # 保存ML模型
        ml_models = {
            "model_L": self.model_L,
            "model_a": self.model_a,
            "model_b": self.model_b,
        }
        with open(out_dir / "ml_models.pkl", "wb") as f:
            pickle.dump(ml_models, f)
        
        return {
            "rts_params": "rts_params.json",
            "ml_models": "ml_models.pkl",
        }
    
    @classmethod
    def load(cls, model_dir: Path) -> MLResidualModel:
        """加载模型"""
        # 加载RTS参数
        rts_params = json.loads(
            (model_dir / "rts_params.json").read_text(encoding="utf-8")
        )
        
        # 加载ML模型
        with open(model_dir / "ml_models.pkl", "rb") as f:
            ml_models = pickle.load(f)
        
        return cls(
            rts_alpha=np.array(rts_params["alpha"], dtype=np.float64),
            rts_beta=np.array(rts_params["beta"], dtype=np.float64),
            rts_gamma=np.array(rts_params["gamma"], dtype=np.float64),
            material_keys=rts_params["material_keys"],
            n_layers=rts_params["n_layers"],
            layer_names_order=rts_params["layer_names_order"],
            model_L=ml_models["model_L"],
            model_a=ml_models["model_a"],
            model_b=ml_models["model_b"],
            feature_names=rts_params["feature_names"],
```

```python
from oc_xgb.xgb_features import build_layer_sequences, build_gpr_features
from oc_xgb.color_space import rgb01_to_lab

def _build_ml_features(
    recipes: list[dict],
    material_keys: list[str],
    base_lab: np.ndarray,
    n_layers: int,
    layer_names_order: str,
    k1: float,
    k2: float,
    backing: str,
) -> np.ndarray:
    """
    构建ML残差模型的特征
    
    参考 build_gpr_features 的实现
    """
    # 基础特征：材料用量
    X_amounts = np.zeros((len(recipes), len(material_keys)), dtype=np.float32)
    for i, recipe in enumerate(recipes):
        for j, key in enumerate(material_keys):
            X_amounts[i, j] = recipe.get(key, 0.0)
    
    # RTS预测值作为特征
    X_rts_lab = base_lab  # (N, 3)
    
    # 层统计特征
    layer_counts = np.sum(X_amounts > 0, axis=1, keepdims=True)  # 非空层数
    total_amounts = np.sum(X_amounts, axis=1, keepdims=True)
    
    # 拼接所有特征
    X = np.concatenate([
        X_amounts,           # 材料用量
        X_rts_lab,           # RTS预测的Lab
        layer_counts,        # 非空层数
        total_amounts,       # 总用量
    ], axis=1)
    
    feature_names = (
        material_keys 
        + ["rts_L", "rts_a", "rts_b", "layer_count", "total_amount"]
    )
    
    return X, feature_names
```
